# Свойства белка по последовательности
> есть аминокислотная последовательность, нужны точные паспортные характеристики

## Что считается точно

Молекулярная масса, изоэлектрическая точка, коэффициент экстинкции при 280 нм,
GRAVY, аминокислотный состав. Это арифметика по последовательности —
результат воспроизводим и не требует оговорок.

## Код

```python
seq = "MKTAYIAKQRQISFVKSHFSRQLEERLGLIEVQ"   # без пробелов, заглавными

W = {'A':71.0788,'R':156.1875,'N':114.1038,'D':115.0886,'C':103.1388,'E':129.1155,
     'Q':128.1307,'G':57.0519,'H':137.1411,'I':113.1594,'L':113.1594,'K':128.1741,
     'M':131.1926,'F':147.1766,'P':97.1167,'S':87.0782,'T':101.1051,'W':186.2132,
     'Y':163.1760,'V':99.1326}
mw = sum(W[a] for a in seq) + 18.015
eps = seq.count('W')*5500 + seq.count('Y')*1490 + (seq.count('C')//2)*125

H = {'A':1.8,'R':-4.5,'N':-3.5,'D':-3.5,'C':2.5,'E':-3.5,'Q':-3.5,'G':-0.4,'H':-3.2,
     'I':4.5,'L':3.8,'K':-3.9,'M':1.9,'F':2.8,'P':-1.6,'S':-0.8,'T':-0.7,'W':-0.9,
     'Y':-1.3,'V':4.2}
gravy = sum(H[a] for a in seq)/len(seq)

pKa = {'D':3.65,'E':4.25,'C':8.18,'Y':10.07,'H':6.00,'K':10.53,'R':10.76}
def charge(pH):
    z = 1/(1+10**(pH-8.2)) - 1/(1+10**(3.65-pH))     # N- и C-концы
    for a, p in pKa.items():
        n = seq.count(a)
        z += n*(1/(1+10**(pH-p)) if a in 'HKR' else -1/(1+10**(p-pH)))
    return z
lo, hi = 0.0, 14.0
for _ in range(60):
    mid = (lo+hi)/2
    lo, hi = (mid, hi) if charge(mid) > 0 else (lo, mid)
pI = (lo+hi)/2

print(f"длина {len(seq)} а.о., MW {mw:.1f} Да, pI {pI:.2f}, e280 {eps} М⁻¹см⁻¹")
print(f"A280=1.0 соответствует {mw/eps:.2f} мг/мл" if eps else "нет Trp/Tyr — по A280 не количественить")
print(f"GRAVY {gravy:.2f} — {'гидрофобный, риск телец включения' if gravy > 0 else 'гидрофильный'}")
result = {"MW_Да": round(mw,1), "pI": round(pI,2), "e280": eps,
          "мг_мл_при_A280_1": round(mw/eps,2) if eps else None, "GRAVY": round(gravy,2)}
```

## Практические выводы

- **ε280 = 0** (нет Trp/Tyr) — концентрацию по A280 мерить нельзя, нужен
  Брэдфорд или BCA. Сказать об этом сразу.
- **pI близок к рабочему pH** — белок будет агрегировать; буфер сместить
  не менее чем на единицу.
- **GRAVY > 0** — повышенный риск телец включения, планировать рефолдинг
  или понижение температуры индукции.
